### Implementation Approach:

```typescript
// src/mocks/handlers/trading.ts
import { http, HttpResponse, delay } from 'msw'
import { usePaperTradingStore } from '@/lib/stores/paper-trading-store'

export const tradingHandlers = [
  // POST /api/trades - Execute trade
  http.post('/api/trades', async ({ request }) => {
    await delay(Math.random() * 300 + 200) // 200-500ms
    const trade = await request.json()

    // Validate trade
    const store = usePaperTradingStore.getState()
    const cost = trade.quantity * trade.price

    if (cost > store.balance) {
      return HttpResponse.json(
        { error: 'Insufficient balance' },
        { status: 400 }
      )
    }

    // Execute trade (updates Zustand store)
    store.executeTrade(trade, trade.price)

    return HttpResponse.json({
      success: true,
      trade: {
        id: `trade-${Date.now()}`,
        ...trade,
        executedAt: new Date().toISOString()
      }
    })
  }),

  // GET /api/trades - Trade history
  http.get('/api/trades', async () => {
    await delay(100)
    const trades = usePaperTradingStore.getState().trades

    return HttpResponse.json({
      trades,
      total: trades.length
    })
  }),

  // GET /api/positions - Active positions
  http.get('/api/positions', async () => {
    await delay(100)
    const positions = usePaperTradingStore.getState().positions

    return HttpResponse.json({
      positions,
      total: positions.length
    })
  }),

  // DELETE /api/positions/:id - Close position
  http.delete('/api/positions/:id', async ({ params }) => {
    await delay(150)
    const store = usePaperTradingStore.getState()
    store.closePosition(params.id)

    return HttpResponse.json({ success: true })
  })
]
```
