```typescript
// src/mocks/handlers/opportunities.ts
import { http, HttpResponse, delay } from 'msw'
import { mockOpportunities, mockAIRecommendations } from '@/mocks/data'

export const opportunitiesHandlers = [
  http.get('/api/opportunities', async () => {
    await delay(200)
    return HttpResponse.json({
      opportunities: mockOpportunities,
      total: mockOpportunities.length
    })
  }),

  http.get('/api/ai/recommendations', async ({ request }) => {
    await delay(1000) // Simulate AI processing time
    const url = new URL(request.url)
    const instrumentId = url.searchParams.get('instrumentId')
    const provider = url.searchParams.get('provider') || 'openai'

    return HttpResponse.json({
      provider,
      recommendation: {
        action: 'buy', // buy/sell/hold
        confidence: 0.75,
        reasoning: 'Mock AI analysis: Technical indicators suggest upward momentum...',
        targetPrice: 150.00,
        stopLoss: 140.00,
        timestamp: new Date().toISOString()
      }
    })
  }),

  http.get('/api/ai/screenshots', async () => {
    await delay(150)
    return HttpResponse.json({
      screenshots: [
        {
          id: 'ss-1',
          instrumentId: 'inst-1',
          timestamp: Date.now() - 3600000,
          analysis: 'Bullish pattern detected',
          provider: 'openai'
        }
      ]
    })
  })
]
```
