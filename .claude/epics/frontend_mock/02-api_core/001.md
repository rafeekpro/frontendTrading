```typescript
// src/mocks/handlers/instruments.ts
import { http, HttpResponse, delay } from 'msw'
import { mockInstruments } from '@/mocks/data'
import { generateCandlesticks } from '@/lib/mock-data/generators'

export const instrumentHandlers = [
  // GET /api/instruments - List all instruments
  http.get('/api/instruments', async () => {
    await delay(150) // Realistic latency
    return HttpResponse.json({
      instruments: mockInstruments,
      total: mockInstruments.length,
      timestamp: Date.now()
    })
  }),

  // GET /api/instruments/:id - Single instrument
  http.get('/api/instruments/:id', async ({ params }) => {
    await delay(100)
    const instrument = mockInstruments.find(i => i.id === params.id)

    if (!instrument) {
      return HttpResponse.json(
        { error: 'Instrument not found' },
        { status: 404 }
      )
    }

    return HttpResponse.json(instrument)
  }),

  // GET /api/instruments/:id/candlesticks
  http.get('/api/instruments/:id/candlesticks', async ({ params, request }) => {
    await delay(200)
    const url = new URL(request.url)
    const timeframe = url.searchParams.get('timeframe') || '1h'

    const validTimeframes = ['1m', '5m', '15m', '1h', '4h', '1d']
    if (!validTimeframes.includes(timeframe)) {
      return HttpResponse.json(
        { error: 'Invalid timeframe' },
        { status: 400 }
      )
    }

    const instrument = mockInstruments.find(i => i.id === params.id)
    if (!instrument) {
      return HttpResponse.json(
        { error: 'Instrument not found' },
        { status: 404 }
      )
    }

    const candlesticks = generateCandlesticks(
      instrument.currentPrice,
      100,
      timeframe
    )

    return HttpResponse.json({
      instrumentId: params.id,
      timeframe,
      data: candlesticks
    })
  })
]
```
